Vous souhaitez effectuer un tirage de secours en toute sécurité sur notre tyrolienne, réservez un créneau sur notre calendrier en allant sur https://bit.ly/pps-rdv, en cliquant sur le bouton, ou en sélectionnant directement un créneau ci-dessous :

<!-- Google Calendar Appointment Scheduling begin -->

<script>
(function() {
  var target = document.currentScript;
  window.addEventListener('load', function() {
    calendar.schedulingButton.load({
      url: 'https://calendar.google.com/calendar/appointments/AcZssZ1Ho6PmWyPwpEIFqJVTjZNGyNjQihabbGz1PtA=?gv=true',
      color: '#039BE5',
      label: 'Réserver un rendez-vous',
      target,
    });
  });
})();
</script>
